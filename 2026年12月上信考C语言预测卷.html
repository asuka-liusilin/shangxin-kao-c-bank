第 13 题9 分

统计指定范围内的偶数

输入整数 lo、hi（0 ≤ lo ≤ hi ≤ 1000），统计闭区间 [lo, hi] 内所有偶数的个数和总和。程序有 3 处错误。 请写出各处修改后的语句。

#include <stdio.h>
int main(void) {
    int lo, hi, x, count = 0, sum = 0;
    scanf("%d%d", &lo, &hi);
    for (x = lo; x < hi; ++x) {
        if (x % 2 == 1) {
            ++count;
            sum = x;
        }
    }
    printf("%d %d\n", count, sum);
    return 0;
}
样例输入

样例输出
3 12
查看参考答案与解析
  1. x < hi 改为 x <= hi。上端点也要检查。
  2. x % 2 == 1 改为 x % 2 == 0。偶数除以 2 的余数为 0。
  3. sum = x; 改为 sum += x;。在原有总和上累加。

清理标签字符串

输入一个长度 1～60 的标签，只含英文字母、数字和连字符。删除全部连字符，保持其余字符的相对顺序。若结果为空，输出 EMPTY。程序有 3 处错误。 请写出各处修改后的语句。

#include <stdio.h>
int main(void) {

第 15 题12 分

查找用电量最大的设备

输入设备数 n（1～30），随后每行给出设备编号（1～10 个非空白字符）、功率（整数瓦）和当天运行时间（整数小时）。以功率乘时间得到瓦时，乘积不超过 30000。输出用电量最大设备的编号及用电量；并列时取最早输入者。程序有 4 处错误。 请写出各处修改后的语句。

#include <stdio.h>
typedef struct {
    char id[11];
    int power, hours, energy;
} Device;
int main(void) {
    Device a[30];
    int n, i, best = 0;
    scanf("%d", &n);
    for (i = 0; i < n; ++i) {
        scanf("%10s%d%d", a[i].id, &a[i].power, &a[i].hours);
        a[i].energy = a[i].power + a[i].hours;
    }
    for (i = 1; i < n - 1; ++i) {
        if (a[i].energy >= a[best].energy) best = i;
    }
    printf("%s %d\n", a[0].id, a[best].energy);
    return 0;
}
样例输入
4
A1 100 2
B2 80 5
C3 200 2
D4 90 6
样例输出
D4 540
查看参考答案与解析
  1. a[i].power + a[i].hours 改为 a[i].power * a[i].hours。按定义计算瓦时。
  2. i < n - 1 改为 i < n。最后一条记录也参与比较。
  3. a[i].energy >= a[best].energy 改为 a[i].energy > a[best].energy。相等时保留先前记录。
  4. a[0].id, a[best].energy 改为 a[best].id, a[best].energy。编号和用电量都应取自同一条记录。

第 16 题15 分

最长连续达标天数

输入天数 n（1～100）和目标步数 t（1～30000），再输入 n 个整数，表示每天的步数（0～30000）。步数达到或超过 t 即达标。输出最长连续达标天数及该段的起始日（从 1 开始），以一个空格分隔。若并列，取起始日最早的一段；若没有达标日，输出 0 0。

样例输入
8 6000
6000 6500 2000 7000 6100 3000 8000 9000
样例输出
2 1
查看参考答案与解析

逐日更新当前连续长度。遇到未达标日就清零，仅当长度严格超过历史最大值时记录起点。

#include <stdio.h>
int main(void) {
    int n, target, steps, day, run = 0, best = 0, start = 0;
    scanf("%d%d", &n, &target);
    for (day = 1; day <= n; ++day) {
        scanf("%d", &steps);
        if (steps >= target) {
            ++run;
            if (run > best) {
                best = run;
                start = day - run + 1;
            }
        } else {
            run = 0;
        }
    }
    printf("%d %d\n", best, start);
    return 0;
}

读入并正确判断达标 3 分，连续长度更新 5 分，最长段及并列处理 5 分，无达标及输出格式 2 分。

第 17 题20 分

按站点汇总借车记录

第一行输入记录数 n（1～100）。随后 n 行各含站点编号和借车次数，由空格分隔。编号为 1～8 个大写英文字母或数字，次数为 0～200。同一编号可能出现多次。按站点累计次数，再按总次数从高到低逐行输出“编号 总次数”。总次数相同，按编号的字符串字典序升序输出（按 strcmp 比较）。

样例输入
6
B2 6
A1 4
C3 9
A1 5
B2 2
D4 0
样例输出
A1 9
C3 9
B2 8
D4 0
查看参考答案与解析

用结构体数组保存各站总数。每读一行先按编号查找，再新增或累加。排序时先比较总数，再比较编号，交换时移动整条记录。

#include <stdio.h>
#include <string.h>
typedef struct {
    char id[9];
    int total;
} Station;
int main(void) {
    Station a[100];
    int n, used = 0, i, j, rides;
    char id[9];
    scanf("%d", &n);
    for (i = 0; i < n; ++i) {
        scanf("%8s%d", id, &rides);
        for (j = 0; j < used; ++j)
            if (strcmp(a[j].id, id) == 0) break;
        if (j == used) {
            strcpy(a[j].id, id);
            a[j].total = 0;
            ++used;
        }
        a[j].total += rides;
    }
    for (i = 0; i < used; ++i) {
        for (j = i + 1; j < used; ++j) {
            if (a[j].total > a[i].total ||
                (a[j].total == a[i].total && strcmp(a[j].id, a[i].id) < 0)) {
                Station tmp = a[i];
                a[i] = a[j];
                a[j] = tmp;
            }
        }
        printf("%s %d\n", a[i].id, a[i].total);
    }
    return 0;
}

读入并保存记录 3 分，按编号合并及累加 7 分，按总数降序 4 分，同分按编号升序 4 分，逐行输出 2 分。
